when I receive [AI Turn]
    if <(Is Clone) = (1)> then
        stop [this script]
    end
    if <not <<(Game Running) = (1)> and <(Turn) > ((Game Players) - (Game AI))>>> then
        stop [this script]
    end
    set [AI Busy] to (1)
    set [ai seat] to (Turn)
    set [ai name] to (item (Turn) of [Player Names])
    set [AI Log] to (join (ai name) [ is taking a turn...])
    set [AI Log Card] to (0)
    set [ai action] to [go]
    repeat until <<<<<(ai action) = [stop]> or <(Actions Left) < (1)>> or <(Game Running) = (0)>> or <not <(Turn) = (ai seat)>>> or <(Game Over) = (1)>>
        wait until <<(Button Mode) = []> or <(Game Running) = (0)>>
        if <(Watch Mode) = (1)> then
            wait (1.1) seconds
        end
        call ai think
        call ai act
        if <(ai action) = [special]> then
            set [Selected Slot] to (ai k)
            set [Selected Card] to (ai c)
            broadcast [AI Special] and wait
        end
        broadcast [Resolve Now] and wait
        if <(ai action) = [debt]> then
            wait until <<(Button Mode) = []> or <(Game Running) = (0)>>
            set [Debt Player] to (Turn)
            broadcast [Debt Resolve] and wait
            if <not <(Debt Done) = (1)>> then
                set [ai action] to [stop]
            end
        end
        if <not <(ai action) = [stop]>> then
            if <(Sound Effects) = (1)> then
                start sound [card]
            end
            wait (0) seconds
        end
    end
    if <<(Game Running) = (1)> and <(Turn) = (ai seat)>> then
        set [AI Log] to (join (ai name) [ ended their turn])
        set [AI Busy] to (0)
        set [AI Log] to []
        set [AI Log Card] to (0)
        broadcast [Next Turn]
    else
        set [AI Busy] to (0)
    end

define win check
    if <<(Game Running) = (1)> and <(Game Over) = (0)>> then
        set [wn ms] to (0)
        set [wn q] to (1)
        repeat (Game Players)
            call sets of (wn q)
            if <(Sets N) > (wn ms)> then
                set [wn ms] to (Sets N)
            end
            change [wn q] by (1)
        end
        set [wn end] to (0)
        if <<<(Ending) = (1)> or <(Ending) = (3)>> and <not <(wn ms) < (3)>>> then
            set [wn end] to (1)
            set [Win Reason] to [Someone built 3 sets of 3 walls.]
        end
        if <<<(Ending) = (2)> or <(Ending) = (3)>> and <(Deck Done) = (1)>> then
            set [wn end] to (1)
            set [Win Reason] to [The deck ran out and the final turns are done.]
        end
        if <<<(Ending) = (4)> and <(Deck Done) = (1)>> and <not <(wn ms) < (3)>>> then
            set [wn end] to (1)
            set [Win Reason] to [Both the deck and 3 sets are done.]
        end
        if <(Final Done) = (1)> then
            set [Final Done] to (0)
            if <(Final Kind) = [sets]> then
                set [wn end] to (1)
                set [Win Reason] to (Final Reason)
            end
            if <(wn end) = (1)> then
                call game over
            end
        else
            if <<(wn end) = (1)> and <(Final Round) = (0)>> then
                set [Final Round] to (1)
                set [Final Stop] to (Turn)
                set [Final Kind] to [sets]
                set [Final Reason] to (Win Reason)
                call say (join [Final round! ] (Win Reason)) [Everyone else gets one final turn.] card (0)
            end
        end
    end
